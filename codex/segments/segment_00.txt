  (function ensurePreloadMetadata(){
    const detectors = [
      { test: /\.(css)(\?|#|$)/i, value: 'style' },
      { test: /\.(js)(\?|#|$)/i, value: 'script' },
      { test: /\.(woff2?|ttf|otf|eot)(\?|#|$)/i, value: 'font' },
      { test: /\.(jpe?g|png|gif|webp|avif|svg)(\?|#|$)/i, value: 'image' },
      { test: /\.(mp4|webm|ogv)(\?|#|$)/i, value: 'video' },
      { test: /\.(mp3|wav|ogg)(\?|#|$)/i, value: 'audio' }
    ];

    const applyMetadata = (links) => {
      links.forEach(link => {
        if (!link || link.rel !== 'preload') return;
        if (link.hasAttribute('as') && link.getAttribute('as')) return;

        const href = link.getAttribute('href') || '';
        const match = detectors.find(entry => entry.test.test(href));
        const asValue = match ? match.value : 'fetch';
        link.setAttribute('as', asValue);

        if (asValue === 'font' && !link.hasAttribute('crossorigin')) {
          link.setAttribute('crossorigin', 'anonymous');
        }
      });
    };

    const init = () => applyMetadata(document.head.querySelectorAll('link[rel="preload"]'));
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init, { once: true });
    } else {
      init();
    }

    const observer = new MutationObserver((mutations) => {
      const added = [];
      mutations.forEach(mutation => {
        mutation.addedNodes.forEach(node => {
          if (node.nodeType === 1 && node.tagName === 'LINK' && node.rel === 'preload') {
            added.push(node);
          }
        });
