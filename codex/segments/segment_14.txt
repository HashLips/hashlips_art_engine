}

async function promptLogin() {
  try {
    await bridgeLogin();
    await hydrateMemberState();
    await refreshWantedPosts();
    showNotice('You are now logged in.', { title: 'Login successful', variant: 'success', autoCloseMs: 2200 });
    return true;
  } catch (error) {
    if (error?.message === 'Canceled' || error?.status === 401) {
      return false;
    }
    console.error('promptLogin error', error);
    showNotice(error?.message || 'Unable to log in right now.', {
      title: 'Login failed',
      variant: 'error',
      autoCloseMs: 2600,
    });
    return false;
  }
}

async function performLogout() {
  try {
    await bridgeLogout();
  } catch (error) {
    console.warn('logout error', error);
  } finally {
    await hydrateMemberState();
    await refreshWantedPosts();
    showNotice('You have been logged out.', { title: 'Logged out', variant: 'info', autoCloseMs: 2000 });
  }
}
