    ownerId: doc.ownerId || doc.owner || doc._owner || null,
    wallets,
    contactLine: sanitizePlain(doc.contactLine || ''),
    contactWhatsApp: sanitizePlain(doc.contactWhatsApp || ''),
    contactEmail: sanitizeEmail(doc.contactEmail || ''),
    contactPhone: sanitizePhone(doc.contactPhone || ''),
    contactTelegram: sanitizePlain(doc.contactTelegram || ''),
    contactMessage: sanitizeRich(doc.contactMessage || doc.message || ''),
    tokenId,
    traitQuery: sanitizePlain(doc.traitQuery || ''),
    createdAt: doc.createdAt || doc._createdDate || null,
    updatedAt: doc.updatedAt || doc._updatedDate || null,
    descriptionRich: sanitizeRich(doc.description || ''),
    ownerDisplayName: sanitizePlain(doc.ownerDisplayName || '')
  };
  const memberId = memberState.id || null;
  mapped.isOwner = Boolean(memberId && mapped.ownerId && String(mapped.ownerId) === String(memberId));
  wantedState.index.set(id, mapped);
  return mapped;
}

function mergeWantedIntoListings(auctions){
  const auctionList = Array.isArray(auctions) ? auctions : LISTINGS.filter((item) => item.kind !== 'wanted');
  const wantedListings = wantedState.list.slice();
  LISTINGS = [...auctionList, ...wantedListings];
}

  return listWalletConnections().map(({ type, chain, address, verified, signature, message }) => ({
    type,
    chain,
    address,
    verified,
    signature,
    message
  }));
}

async function refreshWantedPosts({ silent } = {}){
  try{
    const allPromise = listWantedPosts().catch((err) => {
      console.warn('listWantedPosts error', err);
      return [];
    });
    const minePromise = AUTH.isLoggedIn
      ? getMyWantedPosts().catch((err) => {
          console.warn('getMyWantedPosts error', err);
          return [];
        })
      : Promise.resolve([]);
    const [allDocs, myDocs] = await Promise.all([allPromise, minePromise]);
    wantedState.index.clear();
    wantedState.list = Array.isArray(allDocs) ? allDocs.map(mapWantedDocument).filter(Boolean) : [];
    wantedState.mine = Array.isArray(myDocs) ? myDocs.map(mapWantedDocument).filter(Boolean) : [];
    if (!silent){
      renderWantedListings();
    }
    return wantedState.list;
  } catch (err){
    console.error('refreshWantedPosts error:', err);
    if (!silent){
      showNotice(err?.message || 'Unable to load wanted posts.', { title:'Request failed', variant:'error', autoCloseMs:3200 });
    }
    return [];
  }
}
