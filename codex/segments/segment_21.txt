function ensureAuth(actionLabel, opts = {}){
  const requiresWallet = Boolean(opts.requiresWallet);
  if (!AUTH.isLoggedIn){
    showNotice(`Please log in first${actionLabel ? ` to ${actionLabel}` : ''}.`, { title:'Login required', variant:'error', autoCloseMs:2200 });
    openLoginModal();
    return false;
  }
  if (requiresWallet && !hasEvmWallet()){
    showNotice('Connect an EVM wallet first.', { title:'Wallet required', variant:'error', autoCloseMs:2200 });
    openLoginModal();
    return false;
  }
  return true;
}

function computeTraitStats(){
  const stats = {}; TOTAL_SUPPLY = TOKENS.length;
  for(const t of TOKENS){
    for(const [k,v0] of Object.entries(t.traits||{})){
      const kStr = String(k); const v = String(v0 ?? "");
      if(!stats[kStr]) stats[kStr] = { __total:0, values:{} };
      const node = stats[kStr]; node.__total++; node.values[v] = (node.values[v]||0) + 1;
    }
  }
  TRAIT_STATS = stats;
}
function traitOccurrence(k,v){
  if(!TRAIT_STATS||!k) return {count:0,pct:0};
  const node=TRAIT_STATS[k]; if(!node) return {count:0,pct:0};
  const c=node.values[String(v)]||0; const pct=TOTAL_SUPPLY ? (c/TOTAL_SUPPLY*100) : 0;
  return {count:c,pct};
}
