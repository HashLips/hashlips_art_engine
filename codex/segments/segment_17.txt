function walletSignatureMessage(address, chain){
  return `DIKPIK Bazaar wallet verification\nChain: ${chain}\nAddress: ${address}\nTimestamp: ${new Date().toISOString()}`;
}

async function requestEvmSignature(provider, address){
  const message = walletSignatureMessage(address, 'evm');
  try{
    if (!provider?.request) throw new Error('Missing provider interface');
    const signature = await provider.request({
      method: 'personal_sign',
      params: [message, address]
    });
    return { signature, message };
  } catch (err){
    if (err?.code === 4001){
      showNotice('Signature request was cancelled. You can retry later from Wallets.', { title:'Signature skipped', variant:'warn', autoCloseMs:2400 });
    } else {
      console.warn('requestEvmSignature error', err);
    }
    return { signature: null, message };
  }
}
