      const tokenId = l.tokenId != null ? l.tokenId : '';
      const trait = l.traitQuery || '';
      const contactParts = [];
      if (l.contactLine) contactParts.push(`Line: ${l.contactLine}`);
      if (l.contactWhatsApp) contactParts.push(`WhatsApp: ${l.contactWhatsApp}`);
      if (l.contactTelegram) contactParts.push(`Telegram: ${l.contactTelegram}`);
      if (l.contactEmail) contactParts.push(`Email: ${l.contactEmail}`);
      if (l.contactPhone) contactParts.push(`Phone: ${l.contactPhone}`);
      if (l.contactMessage) contactParts.push(l.contactMessage);
      const contact = contactParts.join(' | ');
      return [
        l.id || l._id || '',
        l.kind || '',
        l.title ?? '',
        price,
        currency,
        tokenId,
        trait,
        l.notes ?? '',
        contact,
        createdISO,
        updatedISO,
        l.min ?? '',
        l.min_ccy ?? '',
        l.escrowed ? 'true' : 'false'
      ];
    });
    const csv = [header, ...rows].map(r => r.map(csvEscape).join(",")).join("\n");
    downloadText("dikpik_bazaar_listings.csv", csv);
  });
}
