    let saved;
    if (id){
      saved = await updateWantedPost(id, normalized);
      showNotice('Wanted post updated.', { title:'Post updated', variant:'success', autoCloseMs:2200 });
    } else {
      saved = await createWantedPost(normalized);
      showNotice('Wanted post created.', { title:'Post created', variant:'success', autoCloseMs:2200 });
    }
    if (saved) mapWantedDocument(saved);
    await refreshWantedPosts({ silent: false });
    renderListings();
    resetWantedForm();
  } catch (err){
    console.error('submitWantedForm error:', err);
    showNotice(err?.message || 'Failed to save wanted post.', { title:'Request failed', variant:'error', autoCloseMs:3200 });
  }
}

function syncRelayInputs(){
  if($('os_chain')) $('os_chain').value = OS_SETTINGS.chain || 'ethereum';
  if($('os_contract')) $('os_contract').value = OS_SETTINGS.contract || '';
}

async function hydrateRelaySettings(){
  if (!HAS_RELAY_API) {
    syncRelayInputs();
    return;
  }
  const remote = await loadRelaySettingsRemote();
  if(remote){
    OS_SETTINGS.chain = remote.chain || OS_SETTINGS.chain;
    OS_SETTINGS.contract = remote.contract || OS_SETTINGS.contract;
    cacheOS();
  }
  syncRelayInputs();
}

async function persistRelaySettings(){
  OS_SETTINGS.chain = ($('os_chain')?.value || 'ethereum').trim() || 'ethereum';
  OS_SETTINGS.contract = ($('os_contract')?.value || '').trim();
  cacheOS();
  if (!HAS_RELAY_API){
    showNotice('Backend unavailable. Settings saved locally.', { title: 'Offline draft', variant: 'warning', autoCloseMs: 2600 });
    syncRelayInputs();
    return;
  }
  try{
    const saved = await saveRelaySettingsRemote(OS_SETTINGS);
    OS_SETTINGS.chain = saved.chain || OS_SETTINGS.chain;
    OS_SETTINGS.contract = saved.contract || OS_SETTINGS.contract;
    cacheOS();
    showNotice("OpenSea settings saved.", { title: "Saved", variant: "success", autoCloseMs: 1800 });
  }catch(err){
    showNotice(`Failed to save settings: ${err?.message || err}`, { title: "Save failed", variant: "error", autoCloseMs: 2600 });
  }
  syncRelayInputs();
  renderListings();
}
