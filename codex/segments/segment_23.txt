  if(!s) return "Associate";
  if(/(^|\b)(don|godfather|boss)\b/.test(s)) return "Boss (Don)";
  if(/\bunder[-\s]?boss\b/.test(s)) return "Underboss";
  if (/\bconsig(liere)?\b|\badvis(or|er)\b/.test(s)) return "Consigliere";
  if(/\bcapo(regime|rgime)?\b|\bcapo\b|\bcaptain\b/.test(s)) return "Caporegime";
  if(/\bsold(ier|ato)|\bfoot|\bgrunt\b/.test(s)) return "Soldier";
  if(/\bassociat|affiliate|runner|associate\b/.test(s)) return "Associate";
  return "Associate";
}
function getRankText(t){ if(t._canonRank) return t._canonRank; const explicit = t?.traits?.Rank ?? t?.traits?.role ?? t?.traits?.ROLE ?? ""; return normalizeToCanon(explicit); }

function canonOccurrence(label){ const c = (CANON_STATS && CANON_STATS[label]) || 0; const pct = CANON_TOTAL ? (c/CANON_TOTAL*100) : 0; return {count:c,pct}; }

/* ================== Canon assignment ================== */
function applyMafiaCanon(){
  for(const [family,list] of FACTIONS.entries()){
    const n = list.length; if(!n) continue;
    let boss = n >= 1 ? 1 : 0, underboss = n >= 3 ? 1 : 0, consigliere = n >= 2 ? 1 : 0;
    let remaining = n - (boss + underboss + consigliere), capos = 0, soldiers = 0, associates = 0;
    if (remaining > 0){
      capos = Math.round(n*0.10); if(n>=12) capos = Math.max(capos,4); capos = Math.min(capos,10,remaining); remaining -= capos;
      soldiers = Math.round(n*0.30); soldiers = Math.min(soldiers,remaining); remaining -= soldiers;
      associates = Math.max(0,remaining);
    }
    const byScore = [...list].sort((a,b)=> (b.rarityScore??-1e9)-(a.rarityScore??-1e9) || (b.nftScore??-1e9)-(a.nftScore??-1e9) || a.tokenId-b.tokenId);
    const assigned = new Set();
    const prefer = label => t => normalizeToCanon(t?.traits?.Rank ?? t?.traits?.Role) === label;
    const take = (f,c) => { const out = []; for(const t of byScore){ if(assigned.has(t)) continue; if(f(t)){ out.push(t); assigned.add(t); if(out.length===c) break; } } return out; };
    const takeAny = c => { const out = []; for(const t of byScore){ if(assigned.has(t)) continue; out.push(t); assigned.add(t); if(out.length===c) break; } return out; };
    const b1 = take(prefer("Boss (Don)"), boss); b1.forEach(t=>t._canonRank="Boss (Don)"); if(b1.length < boss) takeAny(boss - b1.length).forEach(t=>t._canonRank="Boss (Don)");
    const u1 = take(prefer("Underboss"), underboss); u1.forEach(t=>t._canonRank="Underboss"); if (u1.length < underboss) takeAny(underboss - u1.length).forEach(t=>t._canonRank="Underboss");
    const c1 = take(prefer("Consigliere"), consigliere); c1.forEach(t=>t._canonRank="Consigliere"); if (c1.length < consigliere) takeAny(consigliere - c1.length).forEach(t=>t._canonRank="Consigliere");
    const capPicked = take(prefer("Caporegime"), capos); capPicked.forEach(t=>t._canonRank="Caporegime"); if(capPicked.length<capos) takeAny(capos-capPicked.length).forEach(t=>t._canonRank="Caporegime");
    const solPicked = take(prefer("Soldier"), soldiers); solPicked.forEach(t=>t._canonRank="Soldier"); if(solPicked.length<soldiers) takeAny(soldiers-solPicked.length).forEach(t=>t._canonRank="Soldier");
    for(const t of byScore){ if(!assigned.has(t)){ t._canonRank="Associate"; assigned.add(t); } }
  }
}
